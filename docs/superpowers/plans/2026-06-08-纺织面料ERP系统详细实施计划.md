## 一、实施计划概览

| 成果 | 说明 | 优先级 |
|------|------|--------|
| 后端功能完善 | 155个功能点全部实现 | P0 |

import { Injectable } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { InventoryBatch } from '../entities/inventory-batch.entity';
import { CodeGenerator } from '../../../common/utils';

@Injectable()
export class InventoryService {
  constructor(
    @InjectRepository(InventoryBatch)
    private readonly batchRepository: Repository<InventoryBatch>,
  ) {}

  /**
   * 优化的FIFO批次查询
   * 使用索引优化查询性能
   */
  async findAvailableBatches(productId: string, quantity: number) {
    // 使用索引查询，避免全表扫描
    return this.batchRepository.find({
      where: {
        productId,
        status: 'AVAILABLE',
      },
      order: {
        productionDate: 'ASC',  // FIFO：按生产日期排序
        createdAt: 'ASC',       // 次要排序：按创建时间
      },
      // 只查询必要的字段
      select: ['id', 'batchNo', 'rollNo', 'quantity', 'productionDate'],
      // 使用take限制数量，避免一次性查询过多
      take: 100,
    });
  }

  /**
   * 优化的批次库存查询
   */
  async getInventoryByProduct(productId: string) {
    const result = await this.batchRepository
      .createQueryBuilder('batch')
      .select('batch.productId', 'productId')
      .addSelect('batch.status', 'status')
      .addSelect('SUM(batch.quantity)', 'totalQuantity')
      .addSelect('COUNT(batch.id)', 'batchCount')
      .where('batch.productId = :productId', { productId })
      .groupBy('batch.productId')
      .addGroupBy('batch.status')
      .getRawMany();

    return result;
  }
}

// frontend/src/app/core/services/auth.service.ts
import { Injectable, signal, computed } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Router } from '@angular/router';
import { tap, catchError, of } from 'rxjs';
import { environment } from '../../../environments/environment';

export interface LoginRequest {
  username: string;
  password: string;
}

export interface User {
  id: string;
  username: string;
  email: string;
  roles: string[];
}

@Injectable({
  providedIn: 'root',
})
export class AuthService {
  private userSignal = signal<User | null>(null);
  
  readonly user = this.userSignal.asReadonly();
  readonly isAuthenticated = computed(() => !!this.userSignal());
  readonly username = computed(() => this.userSignal()?.username || '');

  constructor(
    private http: HttpClient,
    private router: Router
  ) {
    this.loadUserFromStorage();
  }

  private loadUserFromStorage() {
    const token = localStorage.getItem('token');
    const user = localStorage.getItem('user');
    if (token && user) {
      this.userSignal.set(JSON.parse(user));
    }
  }

  login(credentials: LoginRequest) {
    return this.http.post<{ token: string; user: User }>(
      `${environment.apiUrl}/auth/login`,
      credentials
    ).pipe(
      tap((response) => {
        localStorage.setItem('token', response.token);
        localStorage.setItem('user', JSON.stringify(response.user));
        this.userSignal.set(response.user);
      })
    );
  }

  logout() {
    localStorage.removeItem('token');
    localStorage.removeItem('user');
    this.userSignal.set(null);
    this.router.navigate(['/login']);
  }

  getToken(): string | null {
    return localStorage.getItem('token');
  }
}

        </app-pagination>

export class ProductListComponent implements OnInit {
  private productService = inject(ProductService);

  loading = signal(true);
  products = signal<Product[]>([]);
  
  filters = {
    code: '',
    name: '',
    type: '',
    status: '',
  };

  pagination = signal({
    page: 1,
    limit: 20,
    total: 0,
  });

  ngOnInit() {
    this.loadProducts();
  }

  loadProducts() {
    this.loading.set(true);
    this.productService.getProducts({
      ...this.filters,
      page: this.pagination().page,
      limit: this.pagination().limit,
    }).subscribe({
      next: (response) => {
        this.products.set(response.items);
        this.pagination.update(p => ({
          ...p,
          total: response.total,
        }));
        this.loading.set(false);
      },
      error: () => {
        this.loading.set(false);
      },
    });
  }

  onFilterChange() {
    this.pagination.update(p => ({ ...p, page: 1 }));
    this.loadProducts();
  }

  onPageChange(page: number) {
    this.pagination.update(p => ({ ...p, page }));
    this.loadProducts();
  }

  getTypeLabel(type: string): string {
    const labels: Record<string, string> = {
      'FABRIC': '面料',
      'GREY_CLOTH': '坯布',
      'ACCESSORY': '辅料',
      'DYE': '染料',
      'OTHER': '其他',
    };
    return labels[type] || type;
  }

  getStatusLabel(status: string): string {
    return status === 'ACTIVE' ? '启用' : '停用';
  }

  getStatusClass(status: string): string {
    return status === 'ACTIVE'
      ? 'px-2 inline-flex text-xs leading-5 font-semibold rounded-full bg-green-100 text-green-800'
      : 'px-2 inline-flex text-xs leading-5 font-semibold rounded-full bg-red-100 text-red-800';
  }
}
