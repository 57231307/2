### 1.2 技术分层

## 5. 技术选型详解

### 5.1 前端选型理由

| 技术 | 选型理由 | 替代方案 |
|------|----------|----------|
| Angular 22 | 企业级框架、TypeScript原生、生态完善、长周期支持 | React + Vue |
| SignalForms | 响应式表单、性能优异、TypeScript支持好 | Reactive Forms |
| OnPush | 默认开启，减少变更检测，提升性能 | Default策略 |
| Tailwind CSS | 原子化CSS、开发效率高、可定制性强 | SCSS + Material UI |

### 5.2 后端选型理由

| 技术 | 选型理由 | 替代方案 |
|------|----------|----------|
| NestJS | 模块化架构、TypeScript、与Angular一致、依赖注入 | Spring Boot + Go |
| TypeORM | 与NestJS集成好、TypeScript支持、Active Record模式 | Prisma + Sequelize |
| PostgreSQL | 强大的事务、分区表、JSON支持、企业级特性 | MySQL + MongoDB |
| Redis | 高性能缓存、分布式锁、消息队列、会话存储 | Memcached |

### 5.3 数据库选型理由

```sql
-- 为什么选择 PostgreSQL 的面料行业场景
-- 1. 分区表支持（销售订单按时间分区）
CREATE TABLE sales_orders PARTITION BY RANGE (created_at);

-- 2. JSON字段支持（存储面料属性、颜色配方参数）
ALTER TABLE products ADD COLUMN attributes JSONB;
CREATE INDEX idx_products_attributes ON products USING GIN(attributes);

-- 3. 强大的唯一约束
CREATE UNIQUE INDEX idx_color_formula ON color_formulas (product_id, color_code, version);

-- 4. CTE和窗口函数（复杂报表查询）
WITH monthly_sales AS (
  SELECT date_trunc('month', created_at) as month, sum(amount) as total
  FROM sales_orders
  GROUP BY month
)
SELECT month, total, LAG(total) OVER (ORDER BY month) as prev_month
FROM monthly_sales;
```

## 6. 中间件配置

### 6.1 API网关中间件链

```typescript
// NestJS 中间件配置
@Module({})
export class AppModule {
  configure(consumer: MiddlewareConsumer) {
    consumer
      // 1. 请求日志
      .apply(LoggerMiddleware)
      .forRoutes('*')
      
      // 2. 限流熔断
      .apply(RateLimitMiddleware)
      .forRoutes('*')
      
      // 3. 认证鉴权
      .apply(AuthMiddleware)
      .exclude('/api/auth/login', '/api/auth/register')
      .forRoutes('*')
      
      // 4. CORS处理
      .apply(CorsMiddleware)
      .forRoutes('*');
  }
}
```

### 6.2 Redis 集群配置

```typescript
// Redis 集群配置
@Module({})
export class CacheModule {
  static register(): DynamicModule {
    return {
      module: CacheModule,
      providers: [
        {
          provide: 'REDIS_CLIENT',
          useFactory: async (configService: ConfigService) => {
            const client = createCluster({
              rootNodes: [
                { host: 'redis-1', port: 6379 },
                { host: 'redis-2', port: 6379 },
                { host: 'redis-3', port: 6379 },
              ],
              defaults: {
                password: configService.get('REDIS_PASSWORD'),
              },
            });
            await client.connect();
            return client;
          },
          inject: [ConfigService],
        },
      ],
      exports: ['REDIS_CLIENT'],
    };
  }
}
```

### 6.3 数据库连接池配置

```typescript
// TypeORM 配置
@Module({})
export class DatabaseModule {
  static register(): DynamicModule {
    return TypeOrmModule.forRootAsync({
      useFactory: (configService: ConfigService) => ({
        type: 'postgres',
        host: configService.get('DB_HOST'),
        port: configService.get('DB_PORT'),
        username: configService.get('DB_USERNAME'),
        password: configService.get('DB_PASSWORD'),
        database: configService.get('DB_DATABASE'),
        
        // 连接池配置
        extra: {
          max: 50,
          min: 10,
          idleTimeoutMillis: 30000,
          connectionTimeoutMillis: 2000,
        },
        
        // 自动创建表（开发环境）
        synchronize: configService.get('NODE_ENV') === 'development',
        
        // 日志
        logging: configService.get('NODE_ENV') === 'development',
      }),
      inject: [ConfigService],
    });
  }
}
```
