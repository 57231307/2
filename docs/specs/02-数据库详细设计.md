## 1. 数据库设计原则

### 1.1 设计规范

| 规范 | 说明 |
|------|------|
| 命名规范 | 表名：snake_case，列名：snake_case |
| 主键 | 统一使用 UUID 类型 |
| 时间字段 | 统一使用 TIMESTAMPTZ 类型 |
| 软删除 | 使用 deleted_at 字段，不物理删除 |
| 审计字段 | created_at, updated_at, created_by, updated_by |
| 唯一约束 | 业务主键使用 UNIQUE INDEX |

### 1.2 核心实体定义

```sql
-- 基础实体 - 所有表继承
CREATE TABLE base_entity (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
  created_by UUID,
  updated_by UUID,
  deleted_at TIMESTAMPTZ
);
```

## 4. 基础数据模块表结构

### 4.1 产品表 (products)

```sql
CREATE TABLE erp.products (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  product_no VARCHAR(50) NOT NULL,
  name VARCHAR(200) NOT NULL,
  category_id UUID,
  product_type VARCHAR(50),
  composition VARCHAR(500),
  weight NUMERIC(10, 4),
  weight_unit VARCHAR(20) DEFAULT 'g/m2',
  width NUMERIC(10, 2),
  width_unit VARCHAR(20) DEFAULT 'cm',
  shrinkage NUMERIC(5, 2),
  color_fastness VARCHAR(200),
  description TEXT,
  attributes JSONB,
  status VARCHAR(20) DEFAULT 'active',
  created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
  created_by UUID,
  updated_by UUID,
  deleted_at TIMESTAMPTZ,
  
  UNIQUE (product_no),
  CONSTRAINT fk_category FOREIGN KEY (category_id) REFERENCES erp.product_categories(id)
);

CREATE INDEX idx_products_category ON erp.products(category_id);
CREATE INDEX idx_products_status ON erp.products(status);
CREATE INDEX idx_products_attributes ON erp.products USING GIN(attributes);
COMMENT ON TABLE erp.products IS '产品表（面料产品）';
```

  actual_delivery_date TIMESTAMPTZ,

  department_id UUID,
  total_amount NUMERIC(15, 2) NOT NULL,

  currency VARCHAR(10) DEFAULT 'CNY',
  payment_terms VARCHAR(100),
  shipping_address TEXT,
  billing_address TEXT,

  remark TEXT,
  attributes JSONB,
  created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
  created_by UUID,
  updated_by UUID,

### 11.3 分区维护脚本

```sql
-- 自动创建下一个月的分区函数
CREATE OR REPLACE FUNCTION erp.create_next_month_partition()
RETURNS VOID AS $$
DECLARE
  next_month DATE;
  partition_name TEXT;
  start_date TEXT;
  end_date TEXT;
BEGIN
  next_month := date_trunc('month', CURRENT_DATE + INTERVAL '1 month');
  partition_name := 'sales_orders_' || to_char(next_month, 'yyyy_mm');
  start_date := to_char(next_month, 'yyyy-mm-01');
  end_date := to_char(next_month + INTERVAL '1 month', 'yyyy-mm-01');
  
  EXECUTE format(
    'CREATE TABLE IF NOT EXISTS erp.%I PARTITION OF erp.sales_orders FOR VALUES FROM (%L) TO (%L)',
    partition_name, start_date, end_date
  );
  
  RAISE NOTICE 'Created partition: %', partition_name;
END;
$$ LANGUAGE plpgsql;

-- 归档旧分区函数
CREATE OR REPLACE FUNCTION erp.archive_old_partitions(months_ago INTEGER DEFAULT 24)
RETURNS VOID AS $$
DECLARE
  cutoff_date DATE;
  partition_record RECORD;
BEGIN
  cutoff_date := date_trunc('month', CURRENT_DATE - INTERVAL '1 month' * months_ago);
  
  FOR partition_record IN 
    SELECT inhrelid::regclass AS partition_name
    FROM pg_inherits
    WHERE inhparent = 'erp.sales_orders'::regclass
      AND inhrelid::text < ('erp.sales_orders_' || to_char(cutoff_date, 'yyyy_mm'))
  LOOP
    EXECUTE format(
      'ALTER TABLE %I NO INHERIT erp.sales_orders',
      partition_record.partition_name
    );
    RAISE NOTICE 'Archived partition: %', partition_record.partition_name;
  END LOOP;
END;
$$ LANGUAGE plpgsql;
```
