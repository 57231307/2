## 二、表结构设计

### 2.1 基础数据表

#### 2.1.1 产品表 (products)

```sql
CREATE TABLE products (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    code VARCHAR(50) NOT NULL UNIQUE,
    name VARCHAR(200) NOT NULL,
    type VARCHAR(20) NOT NULL DEFAULT 'FABRIC',
    spec TEXT,
    unit VARCHAR(20) DEFAULT 'meter',
    color_formula_id UUID,
    pattern_id UUID,
    standard_cost DECIMAL(12, 4),
    sale_price DECIMAL(12, 4),
    status VARCHAR(20) DEFAULT 'ACTIVE',
    images JSONB DEFAULT '[]',
    attachments JSONB DEFAULT '[]',
    metadata JSONB DEFAULT '{}',
    remark TEXT,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    created_by UUID,
    updated_by UUID
);

-- 索引
CREATE INDEX idx_products_code ON products(code);
CREATE INDEX idx_products_type ON products(type);
CREATE INDEX idx_products_status ON products(status);
CREATE INDEX idx_products_name ON products(name);
CREATE INDEX idx_products_color_formula ON products(color_formula_id);
CREATE INDEX idx_products_pattern ON products(pattern_id);

-- 约束
ALTER TABLE products ADD CONSTRAINT chk_products_type 
    CHECK (type IN ('FABRIC', 'GREIGE_GOODS', 'AUXILIARY', 'DYE', 'OTHER'));
ALTER TABLE products ADD CONSTRAINT chk_products_status 
    CHECK (status IN ('ACTIVE', 'INACTIVE', 'DISCONTINUED'));
```

**字段说明：**

| 字段名 | 类型 | 约束 | 说明 |
|--------|------|------|------|
| id | UUID | PK | 主键 |
| code | VARCHAR(50) | NOT NULL, UNIQUE | 产品编码 |
| name | VARCHAR(200) | NOT NULL | 产品名称 |
| type | VARCHAR(20) | NOT NULL | 产品类型 |
| spec | TEXT | | 规格参数 |
| unit | VARCHAR(20) | DEFAULT 'meter' | 计量单位 |
| color_formula_id | UUID | FK | 关联颜色配方 |
| pattern_id | UUID | FK | 关联花型 |
| standard_cost | DECIMAL(12,4) | | 标准成本 |
| sale_price | DECIMAL(12,4) | | 销售价格 |
| status | VARCHAR(20) | DEFAULT 'ACTIVE' | 状态 |
| images | JSONB | DEFAULT '[]' | 产品图片 |
| attachments | JSONB | DEFAULT '[]' | 附件 |
| metadata | JSONB | DEFAULT '{}' | 扩展数据 |
| remark | TEXT | | 备注 |

## 三、触发器和规则

### 3.1 自动编码生成

```sql
-- 产品编码触发器
CREATE OR REPLACE FUNCTION generate_product_code()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.code IS NULL OR NEW.code = '' THEN
        NEW.code := 'PRD-' || TO_CHAR(CURRENT_DATE, 'YYYYMMDD') || '-' || 
                    LPAD(CAST(
                        COALESCE(
                            (SELECT COUNT(*) + 1 FROM products 
                             WHERE code LIKE 'PRD-' || TO_CHAR(CURRENT_DATE, 'YYYYMMDD') || '-%'),
                            1
                        ) AS VARCHAR
                    ), 4, '0');
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_product_code
    BEFORE INSERT ON products
    FOR EACH ROW
    EXECUTE FUNCTION generate_product_code();
```

### 3.2 库存变动自动记账

```sql
-- 库存变动自动记录台账
CREATE OR REPLACE FUNCTION record_inventory_change()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO inventory_ledger (
        batch_id, product_id, warehouse_id, transaction_type,

        NEW.quantity_change, NEW.quantity_before, NEW.quantity_after,
```

### 4.2 物化视图

```sql
-- 库存汇总物化视图
CREATE MATERIALIZED VIEW inventory_summary AS
SELECT 
    p.id as product_id,
    p.code as product_code,
    p.name as product_name,
    p.type as product_type,
    w.id as warehouse_id,
    w.code as warehouse_code,
    w.name as warehouse_name,
    SUM(b.quantity) as total_quantity,
    SUM(b.available_quantity) as available_quantity,
    COUNT(b.id) as batch_count
FROM inventory_batches b
JOIN products p ON b.product_id = p.id
JOIN warehouses w ON b.warehouse_id = w.id
WHERE b.status = 'ACTIVE'
GROUP BY p.id, p.code, p.name, p.type, w.id, w.code, w.name;

CREATE UNIQUE INDEX idx_inventory_summary 
ON inventory_summary(product_id, warehouse_id);

-- 定时刷新
CREATE OR REPLACE FUNCTION refresh_inventory_summary()
RETURNS void AS $$
BEGIN
    REFRESH MATERIALIZED VIEW CONCURRENTLY inventory_summary;
END;
$$ LANGUAGE plpgsql;
```
